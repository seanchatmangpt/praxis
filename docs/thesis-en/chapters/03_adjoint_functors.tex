\chapter{The Exploration-Exploitation Adjoint Pair}

\section{Base Spaces and Notation}

\begin{defn}[Hypothesis Space and Admission Space]
Let $\Hyp$ be the design hypothesis space (the codomain of the exploration pole): its elements encompass existent symbols, hallucinated APIs, empty tests, and unbuilt specifications. Let $\Adm \subset \Hyp$ be the admitted subspace: elements that compile, run, and whose outputs are captured under the current toolchain. Let $\Rfs = \coprod_{i=1}^{29} \Rfs_i$ be the refusal space---twenty-nine mutually disjoint typed coproduct components.
Crucial structural fact: $\Rfs$ is a coproduct, not a quotient. Refusals preserve their causation and never collapse into a generic error state.
\end{defn}

\begin{defn}[Stratification Function]
The stratification function $\strat : \Hyp \to \{\mathsf{V}, \mathsf{D}, \mathsf{S}, \mathsf{X}, \mathsf{H}, \mathsf{U}\}$ (Verified, Drifted, Stubbed, Dead, Hallucinated, Unverified) is induced by an evidence operator:
\[
\strat(h) = \begin{cases}
\mathsf{V} & \exists\, e \in E_{\text{session}} : \mathsf{compile}(h) \wedge \mathsf{run}(h) = e \\
\mathsf{H} & \nexists\, s \in \mathsf{Sym}(\text{src}) : \mathsf{claim}(h) \mapsto s \\
\mathsf{U} & \text{Otherwise (default exclusion, not default admission)}
\end{cases}
\]
The defect is not the garbage itself, but rather the \textbf{unmarked strata}. When $\strat$ is unapplied, elements in $\Hyp \setminus \Adm$ masquerade as elements in $\Adm$.
\end{defn}

\section{Generation-Verification Adjunction}

\begin{defn}[Adjoint Pair]
Let $G : \Omega \to \Hyp$ be the exploration functor (from the prompt space $\Omega$ generating high-capacity hypotheses), and $V : \Hyp \to \Adm \cup \Rfs$ be the verification functor. They form an approximate Galois connection:
\[
G(\omega) \le_{\Hyp} h \iff \omega \le_{\Omega} V^*(h)
\]
where $V^*$ is the right adjoint of verification. The exploration pole does not require correctness (its value is the coverage measure $\mu(G(\Omega))$); the exploitation pole does not require breadth (its value is the deterministic certainty $\Pr[V \text{ correct}] \to 1$).
\end{defn}

\begin{thm}[Verification Asymmetry]
Let $\cost_G(h)$ be the cost of generating hypothesis $h$, and $\cost_V(h)$ be the cost of deciding $\strat(h)$. In an environment equipped with compilers, probes, and hash comparators:
\[
\cost_V(h) \ll \cost_G(h) \quad \text{and} \quad
\mathrm{Var}[V(h)] \ll \mathrm{Var}[G(\omega)]
\]
\end{thm}

\begin{lem}[Anti-Anchoring Lemma]
Let $\beta(V, \mathrm{auth}(H))$ be the prior bias term of auditor $V$ over hypotheses $H$. Then $\beta(V, H_{\text{self}}) > \beta(V, H_{\text{other}}) \approx 0$. Thus, $G \ne V$ (the separation of author and auditor) is not a division of labor convenience, but a necessary condition for $\Pr[\strat \text{ unbiased}]$. This is the self-referential application of the constitutional decree: "witness is not authority."
\end{lem}
